# OVERVIEW

The Trust Score Dashboard provides a **comprehensive visual analytics view** of a provider's trust profile, including:
- STL score breakdown
- Industry verification strength
- Refilling history
- Complaint impact
- AI trust analysis
- Improvement recommendations

# 8. DASHBOARD DATA STRUCTURE

```typescript
interface TrustDashboard {
  // Overall Score
  overall_score: number;
  level: 'FREE' | 'BASIC' | 'MEDIUM' | 'HIGH' | 'VIP';
  percentile: number;
  
  // Score Breakdown
  breakdown: {
    pss_score: number;
    crb_score: number;
    performance_score: number;
    review_score: number;
    complaint_score: number;
    refilling_score: number;
  };
  
  // Industry Boosts
  industry_boosts: IndustryBoost[];
  
  // Refilling History
  refilling: {
    total: number;
    passed: number;
    failed: number;
    next_due: Date;
  };
  
  // Complaints
  complaints: {
    total: number;
    resolved: number;
    pending: number;
    avg_resolution_days: number;
  };
  
  // AI Analysis
  ai_insights: {
    strengths: string[];
    improvements: string[];
    trend: 'improving' | 'stable' | 'declining';
    trend_points: number;
  };
  
  // Recommendations
  recommendations: Recommendation[];
  
  // Peer Comparison
  peer_comparison: {
    category_average: number;
    percentile: number;
    rank: number;
    total_peers: number;
  };
}
```
